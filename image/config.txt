# Serial UART options for HAT and onboard Bluetooth
#
# All four vendor options are reproducible (board-verified
# 2026-10-15): UART roles are fused into the DTBs at compile time,
# so switching = editing the device_tree line below. Never enable
# the vendor overlay lines (pi3-miniuart-bt / pi3-disable-bt).
#
#   option 0: BT on ttyAMA0/PL011, no HAT UART
#             device_tree=bcm2710-rpi-custom-btpl011.dtb
#             (comment out enable_uart=1 - the firmware keeps the
#             unused mini-uart disabled, stock option-0 semantics)
#   option 1: option 0 + ttyS0 mini-uart for the HAT on GPIO14/15
#             (= /dev/serial0 + console), core clock locked
#             device_tree=bcm2710-rpi-custom-btpl011.dtb
#             enable_uart=1   [core_freq=400: lock at 400 MHz
#                              instead of enable_uart's implicit 250,
#                              preserves vc4/HDMI bandwidth]  << ACTIVE
#   option 2: PL011 on the GPIO header = /dev/serial0 (ttyAMA0,
#             console), BT on the mini-uart @921600 = hci0
#             device_tree=bcm2710-rpi-custom.dtb
#             enable_uart=1   (core-clock lock for the mini-uart BT)
#   option 3: BT off (mini-uart then = plain /dev/ttyS0 on
#             GPIO32/33 under option 2): uncomment smartedge_bt=off
#
# Vendor original (reference): option 0 default / option 1
# enable_uart=1 / option 2 dtoverlay=pi3-miniuart-bt + core_freq=250 /
# option 3 dtoverlay=pi3-disable-bt
#
# NOTE: Only one option active at a time
#################################################################

# --- ACTIVE: vendor option 1 (BT on PL011, HAT mini-uart on the header) ---
device_tree=bcm2710-rpi-custom-btpl011.dtb
enable_uart=1
core_freq=400
# BT baud rate (921600 = proven 2-wire rate; 3M intermittently times out)
dtparam=krnbt_baudrate=921600
# vendor option 3: uncomment to disable onboard BT
#dtparam=smartedge_bt=off

# --- hardware interfaces ---
dtparam=audio=on
camera_auto_detect=1
display_auto_detect=1

# --- display / DRM (vc4 KMS owns HDMI; firmware must not pre-set video=) ---
dtoverlay=vc4-kms-v3d
max_framebuffers=2
disable_fw_kms_setup=1
disable_overscan=1

# --- CPU ---
arm_64bit=1
arm_boost=1
# Avnet GPU firmware + custom revision: default 1200 MHz is board-proven
# (4-core full load ~52C, throttled=0x0). arm_freq=1400 TESTED 2026-10-12:
# firmware fails to apply it, board dies before Linux (HDMI black, no sshd,
# POR loop via ATtiny WDT). DO NOT use.
arm_freq=1200

# --- initramfs ---
auto_initramfs=1
