# CPU frequency ceiling for this board (Avnet GPU firmware + custom revision):
# - commented out / no arm_freq  -> firmware default 1200MHz (board-proven:
#   4-core full load ~52C, throttled=0x0)
# - arm_freq=1400 TESTED 2026-10-12: firmware fails to apply it, board dies
#   before Linux (HDMI black, no sshd, POR loop via ATtiny WDT). DO NOT use.

#################################################################
# Serial UART options for HAT and onboard Bluetooth (6.18 stack)
#
# NOTE: A single DTB covers every case; only option 3 is a switch
#################################################################

### Serial option 0/1/2 (all default): /dev/serial0 = PL011 on the
### GPIO header (ttyAMA0, console), onboard Bluetooth = hci0 on the
### mini-uart (/dev/serial1) @921600
# (compiled into the DTB - no lines needed for the UART roles;
#  enable_uart=1 below locks the core clock the mini-uart BT
#  baud rate depends on)
enable_uart=1

### Serial option 3: Disable onboard bluetooth - the mini-uart then
### shows up as a plain /dev/ttyS0 (GPIO32/33); serial0 stays on the PL011
#dtparam=smartedge_bt=off

### BT baud rate (921600 = proven 2-wire rate; 3M intermittently times out)
dtparam=krnbt_baudrate=921600
